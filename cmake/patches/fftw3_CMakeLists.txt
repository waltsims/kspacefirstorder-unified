cmake_minimum_required(VERSION 3.18)
project(fftw3 C)

include(CheckIncludeFiles)
include(CheckFunctionExists)
include(CheckSymbolExists)
include(CheckTypeSize)

option(ENABLE_FLOAT "Enable single precision" OFF)
option(ENABLE_LONG_DOUBLE "Enable long double precision" OFF)
option(ENABLE_QUAD "Enable quad precision" OFF)
option(BUILD_TESTS "Build tests" OFF)
option(ENABLE_SSE "Enable SSE optimizations" ON)
option(ENABLE_SSE2 "Enable SSE2 optimizations" ON)
option(ENABLE_AVX "Enable AVX optimizations" ON)
option(ENABLE_AVX2 "Enable AVX2 optimizations" ON)

# Platform detection
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(SIZEOF_VOID_P 8)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 4)
    set(SIZEOF_VOID_P 4)
else()
    message(FATAL_ERROR "Unsupported pointer size: ${CMAKE_SIZEOF_VOID_P}")
endif()

# Check for required headers
check_include_files(stddef.h HAVE_STDDEF_H)
check_include_files(stdlib.h HAVE_STDLIB_H)
check_include_files(string.h HAVE_STRING_H)
check_include_files(sys/types.h HAVE_SYS_TYPES_H)
check_include_files(unistd.h HAVE_UNISTD_H)
check_include_files(malloc.h HAVE_MALLOC_H)
check_include_files(alloca.h HAVE_ALLOCA_H)
check_include_files(altivec.h HAVE_ALTIVEC_H)
check_include_files(sys/time.h HAVE_SYS_TIME_H)
check_include_files(time.h HAVE_TIME_H)

# Check for required functions
check_function_exists(gettimeofday HAVE_GETTIMEOFDAY)
check_function_exists(BSDgettimeofday HAVE_BSDGETTIMEOFDAY)
check_function_exists(drand48 HAVE_DRAND48)
check_function_exists(sqrt HAVE_SQRT)
check_function_exists(memset HAVE_MEMSET)

# Configure precision
if(ENABLE_FLOAT)
    set(FFTW_SINGLE 1)
    set(BENCHFFT_SINGLE 1)
    set(FFTW_PREFIX s)
endif()

# Configure SIMD
if(ENABLE_SSE)
    set(HAVE_SSE 1)
endif()
if(ENABLE_SSE2)
    set(HAVE_SSE2 1)
endif()
if(ENABLE_AVX)
    set(HAVE_AVX 1)
endif()
if(ENABLE_AVX2)
    set(HAVE_AVX2 1)
endif()

# Platform-specific settings
if(APPLE)
    set(HAVE_MACOS 1)
endif()
if(UNIX)
    set(HAVE_UNIX 1)
endif()

# Configure files
configure_file(${CMAKE_CURRENT_SOURCE_DIR}/config.h.in ${CMAKE_CURRENT_BINARY_DIR}/config.h)

# Add source files
file(GLOB_RECURSE FFTW_SOURCES 
    ${CMAKE_CURRENT_SOURCE_DIR}/kernel/*.c
    ${CMAKE_CURRENT_SOURCE_DIR}/simd/*.c
    ${CMAKE_CURRENT_SOURCE_DIR}/dft/*.c
    ${CMAKE_CURRENT_SOURCE_DIR}/rdft/*.c
    ${CMAKE_CURRENT_SOURCE_DIR}/reodft/*.c
    ${CMAKE_CURRENT_SOURCE_DIR}/api/*.c
    ${CMAKE_CURRENT_SOURCE_DIR}/support/*.c
    ${CMAKE_CURRENT_SOURCE_DIR}/libbench2/*.c
)

# Create library
if(BUILD_SHARED_LIBS)
    add_library(fftw3f SHARED ${FFTW_SOURCES})
else()
    add_library(fftw3f STATIC ${FFTW_SOURCES})
endif()

target_include_directories(fftw3f PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/api
    ${CMAKE_CURRENT_SOURCE_DIR}/kernel
    ${CMAKE_CURRENT_SOURCE_DIR}/dft
    ${CMAKE_CURRENT_SOURCE_DIR}/rdft
    ${CMAKE_CURRENT_BINARY_DIR}
)

target_compile_definitions(fftw3f PRIVATE
    HAVE_CONFIG_H
    SIZEOF_VOID_P=${SIZEOF_VOID_P}
)

if(UNIX)
    target_link_libraries(fftw3f PUBLIC m)
endif()

# Install
install(TARGETS fftw3f
    RUNTIME DESTINATION bin
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
)
install(FILES api/fftw3.h DESTINATION include)
